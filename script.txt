local function createToggleButton(name, yPos, initialState, callback)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, -20, 0, 28)
    btn.Position = UDim2.new(0, 10, 0, yPos)
    btn.BackgroundColor3 = initialState and Color3.fromRGB(0, 180, 130) or Color3.fromRGB(42, 42, 54)
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamMedium
    btn.TextSize = 11
    btn.Text = name .. " : " .. (initialState and "ON" or "OFF")
    btn.Parent = scrollFrame

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn

    local isOn = initialState

    btn.MouseButton1Click:Connect(function()
        isOn = not isOn
        btn.Text = name .. " : " .. (isOn and "ON" or "OFF")
        btn.BackgroundColor3 = isOn and Color3.fromRGB(0, 180, 130) or Color3.fromRGB(42, 42, 54)
        callback(isOn)
    end)

    if name == "🎯 FLICK MODE" then
        flickBtnReference = btn
    end

    return btn
end

-- Create Touch/Mouse Compatible Sliders
local function createSlider(name, yPos, minVal, maxVal, startVal, callback)
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -20, 0, 16)
    label.Position = UDim2.new(0, 10, 0, yPos)
    label.BackgroundTransparency = 1
    label.Text = name .. ": " .. startVal
    label.TextColor3 = Color3.fromRGB(200, 200, 220)
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 11
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = scrollFrame

    local barBg = Instance.new("TextButton")
    barBg.Size = UDim2.new(1, -20, 0, 12)
    barBg.Position = UDim2.new(0, 10, 0, yPos + 18)
    barBg.BackgroundColor3 = Color3.fromRGB(40, 40, 52)
    barBg.Text = ""
    barBg.Parent = scrollFrame

    local barBgCorner = Instance.new("UICorner")
    barBgCorner.CornerRadius = UDim.new(0, 4)
    barBgCorner.Parent = barBg

    local barFill = Instance.new("Frame")
    barFill.Size = UDim2.new((startVal - minVal)/(maxVal - minVal), 0, 1, 0)
    if name == "⚡ SPEED BOOST" then
        barFill.BackgroundColor3 = Color3.fromRGB(0, 229, 255)
    elseif name == "🦘 JUMP HEIGHT" then
        barFill.BackgroundColor3 = Color3.fromRGB(255, 215, 0)
    elseif name == "🎯 FLICK POWER" then
        barFill.BackgroundColor3 = Color3.fromRGB(255, 180, 0)
    elseif name == "🌫️ FOG DISTANCE" then
        barFill.BackgroundColor3 = Color3.fromRGB(200, 160, 255)
    elseif name == "🔴 RED" then
        barFill.BackgroundColor3 = Color3.fromRGB(255, 80, 80)
    elseif name == "🟢 GREEN" then
        barFill.BackgroundColor3 = Color3.fromRGB(80, 255, 120)
    else
        barFill.BackgroundColor3 = Color3.fromRGB(80, 160, 255)
    end
    barFill.Parent = barBg

    local barFillCorner = Instance.new("UICorner")
    barFillCorner.CornerRadius = UDim.new(0, 4)
    barFillCorner.Parent = barFill

    local dragging = false

    local function updateFromInput(inputPos)
        local mousePos = inputPos.X - barBg.AbsolutePosition.X
        local ratio = math.clamp(mousePos / barBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(minVal + ratio * (maxVal - minVal))
        label.Text = name .. ": " .. val
        barFill.Size = UDim2.new(ratio, 0, 1, 0)
        callback(val)
    end

    barBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            scrollFrame.ScrollingEnabled = false
            updateFromInput(input.Position)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input.Position)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            scrollFrame.ScrollingEnabled = true
        end
    end)
end

-- OLD GUN ANIM SYSTEM
-- ==========================================
local lastUnequippedTime = 0
local animConn1, animConn2, animConn3 = nil, nil, nil

local function setupGunSystem(character)
    local humanoid = character:FindFirstChild("Humanoid")
    local animator = humanoid and humanoid:FindFirstChild("Animator")
    if not animator then return end

    if animConn1 then animConn1:Disconnect() end
    if animConn2 then animConn2:Disconnect() end
    if animConn3 then animConn3:Disconnect() end

    lastUnequippedTime = 0

    if animationEnabled then
        animConn1 = animator.AnimationPlayed:Connect(function(track)
            local tool = character:FindFirstChildOfClass("Tool")
            local isHoldingGun = (tool and tool.Name == TARGET_TOOL_NAME)
            local timeSinceUnequip = os.clock() - lastUnequippedTime
            local withinCooldown = timeSinceUnequip <= COOLDOWN_DURATION
            if isHoldingGun or withinCooldown then
                if track.Priority == Enum.AnimationPriority.Action then
                    track:Stop()
                end
            end
        end)

        animConn2 = character.ChildAdded:Connect(function(child)
            if child:IsA("Tool") and child.Name == TARGET_TOOL_NAME then
                actionSound:Play()
            end
        end)

        animConn3 = character.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") and child.Name == TARGET_TOOL_NAME then
                actionSound:Play()
                lastUnequippedTime = os.clock()
            end
        end)
    end
end

local function setAnimationSystem(enabled)
    animationEnabled = enabled
    if enabled then
        if player.Character then setupGunSystem(player.Character) end
    else
        if animConn1 then animConn1:Disconnect() end
        if animConn2 then animConn2:Disconnect() end
        if animConn3 then animConn3:Disconnect() end
    end
end

local antiFlingLoaded = false

local function loadAntiFling()
    if antiFlingLoaded then return end
    task.spawn(function()
        local success, err = pcall(function()
            loadstring(game:HttpGet('https://raw.githubusercontent.com/Linux6699/DaHubRevival/main/AntiFling.lua'))()
        end)
        if not success then
            warn("Anti-Fling failed to load:", err)
        else
            antiFlingLoaded = true
        end
    end)
end

local function setAntiFling(enabled)
    antiFlingEnabled = enabled
    if enabled then
        loadAntiFling()
    end
end

local function ToggleSystem()
    sistemaAtivo = not sistemaAtivo
    atualizarBotao()
    if humanoid then
        if not sistemaAtivo then
            humanoid.WalkSpeed = defaultWalkSpeed
            humanoid.JumpPower = defaultJumpPower
        end
    end
end

local function ToggleFlick()
    flickEnabled = not flickEnabled
    if flickBtnReference then
        flickBtnReference.Text = "🎯 FLICK MODE : " .. (flickEnabled and "ON" or "OFF")
        flickBtnReference.BackgroundColor3 = flickEnabled and Color3.fromRGB(0, 180, 130) or Color3.fromRGB(42, 42, 54)
    end
end

local function ToggleGUI()
    guiVisible = not guiVisible
    mainPanel.Visible = guiVisible
end

-- TRACK MOBILE/TOUCH SHAKE DELTA
UserInputService.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
        mobileTouchDelta = input.Delta.Magnitude
    end
end)

-- KEYBINDS INPUT LISTENER
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end

    if isRebindingSpeed and input.UserInputType == Enum.UserInputType.Keyboard then
        currentKeybind = input.KeyCode
        isRebindingSpeed = false
        speedKeybindBtn.Text = "🎹 SPEED KEY: " .. currentKeybind.Name
        speedKeybindBtn.BackgroundColor3 = Color3.fromRGB(42, 42, 54)
        return
    end

    if isRebindingFlick and input.UserInputType == Enum.UserInputType.Keyboard then
        flickKeybind = input.KeyCode
        isRebindingFlick = false
        flickKeybindBtn.Text = "🎯 FLICK KEY: " .. flickKeybind.Name
        flickKeybindBtn.BackgroundColor3 = Color3.fromRGB(42, 42, 54)
        return
    end

    if input.KeyCode == currentKeybind then
        ToggleSystem()
    elseif input.KeyCode == flickKeybind then
        ToggleFlick()
    elseif input.KeyCode == Enum.KeyCode.K then
        ToggleGUI()
    end
end)

-- Heartbeat Camera Flick, Jump Power & Speed Multiplier (Cross-Platform)
RunService.Heartbeat:Connect(function()
    if sistemaAtivo and humanoid then
        humanoid.UseJumpPower = true
        humanoid.JumpPower = jumpHeightPower

        local state = humanoid:GetState()
        if state == Enum.HumanoidStateType.Jumping or state == Enum.HumanoidStateType.Freefall then
            
            -- Camera Auto-flick
            if flickEnabled and camera then
                flickDirection = -flickDirection
                local flickAngle = math.rad(flickPower * 0.8) * flickDirection
                camera.CFrame = camera.CFrame * CFrame.Angles(0, flickAngle, 0)
            end

            local delta = UserInputService:GetMouseDelta().Magnitude
            local totalShake = math.max(delta, mobileTouchDelta)
            
            local effectiveShake = flickEnabled and (totalShake + (flickPower * 8)) or totalShake
            local shakeMultiplier = math.clamp(effectiveShake * 2.5, 1, 100)
            
            humanoid.WalkSpeed = speedBoost * shakeMultiplier
            mobileTouchDelta = math.max(0, mobileTouchDelta - 2)
        else
            humanoid.WalkSpeed = defaultWalkSpeed
        end
    end
end)
